   289→   289→  private finalizeRun(won: boolean): void {
   290→   290→    const m = this.meta.data;
   291→   291→    this.coresEarned =
   292→   292→      Math.floor(this.score / 1000) + this.wave * 3 + this.bossKills * 30;
   293→   293→    m.cores += this.coresEarned;
   294→   294→    m.bestScore = Math.max(m.bestScore, Math.floor(this.score));
   295→   295→    m.bestWave = Math.max(m.bestWave, this.wave);
   296→   296→    if (won) {
   297→   297→      m.wins++;
   298→   298→      m.endlessUnlocked = true;
   299→   299→    }
   300→   300→    this.meta.save();

   305→   305→  update(realDt: number): void {
   306→   306→    // decay screen fx always
   307→   307→    this.trauma = Math.max(0, this.trauma - realDt * 3.2);
   308→   308→    this.flashAlpha = Math.max(0, this.flashAlpha - realDt * 2.4);
   309→   309→    if (this.bossBanner) {
   310→   310→      this.bossBanner.t -= realDt;
   311→   311→      if (this.bossBanner.t <= 0) this.bossBanner = null;
   312→   312→    }
   313→   313→    this.mouse.clicked = false;
   314→   314→
   315→   315→    if (this.state !== 'playing') {
   316→   316→      // menus still breathe slightly
   317→   317→      if (this.state === 'title' || this.state === 'perks' || this.state === 'help') {
   318→   318→        this.audio.setIntensity(0.25);
   319→   319→      } else if (this.state === 'paused') {
   320→   320→        this.audio.setIntensity(0.15);
   321→   321→      }
   322→   322→      return;
   323→   323→    }
   324→   324→
   325→   325→    // death cinematic
   326→   326→    let dt = realDt;
   327→   327→    if (this.deathT > 0) {
   328→   328→      this.deathT -= realDt;
   329→   329→      dt = realDt * 0.28;
   330→   330→      if (this.deathT <= 0) {
   331→   331→        this.finalizeRun(false);
   332→   332→        this.state = 'gameover';
   333→   333→        return;
   334→   334→      }
   335→   335→    }
   336→   336→
   337→   337→    // hit-stop freezes world but not fx
   338→   338→    if (this.hitStop > 0) {
   339→   339→      this.hitStop -= realDt;
   340→   340→      return;
   341→   341→    }
   342→   342→
   343→   343→    this.time += dt;
   344→   344→
   345→   345→    // delayed calls
   346→   346→    for (const d of this.delayed) d.t -= dt;
   347→   347→    const ready = this.delayed.filter((d) => d.t <= 0);
   348→   348→    if (ready.length) {
   349→   349→      this.delayed = this.delayed.filter((d) => d.t > 0);
   350→   350→      for (const d of ready) d.fn();
   351→   351→    }
   352→   352→
   353→   353→    updatePlayer(this, dt);
   354→   354→    updateWaves(this, dt);
   355→   355→
   356→   356→    rebuildGrid(this);
   357→   357→    updateEnemies(this, dt);
   358→   358→    separateEnemies(this, dt);
   359→   359→    if (this.boss && this.bossState) {
   360→   360→      updateBoss(this, this.boss, this.bossState, dt);
   361→   361→    }
   362→   362→    fireWeapons(this, dt);
   363→   363→    updateBullets(this, dt);
   364→   364→    updateEBullets(this, dt);
   365→   365→    updateShocks(this, dt);
   366→   366→    sweepDead(this.enemies); // bullets/shocks may have killed enemies
   367→   367→    updateGems(this, dt);
   368→   368→    updateParticles(this, dt);
   369→   369→
   370→   370→    // music intensity follows the fight
   371→   371→    const target =
   372→   372→      this.player.overdrive > 0
   373→   373→        ? 1
   374→   374→        : this.bossActive
   375→   375→          ? 0.85
   376→   376→          : 0.4 + Math.min(0.25, this.wave * 0.016) + Math.min(0.1, this.player.heat / 1000);
   377→   377→    this.audio.setIntensity(target);
   378→   378→
   379→   379→    // contextual hints (first run onboarding)
   380→   380→    this.updateHints(dt);
   381→   381→
   382→   382→    // camera
   383→   383→    const camLead = 0.12;
   384→   384→    const tx = this.player.x + this.player.vx * camLead;
   385→   385→    const ty = this.player.y + this.player.vy * camLead;
   386→   386→    this.camera.x += (tx - this.camera.x) * Math.min(1, 6 * realDt);
   387→   387→    this.camera.y += (ty - this.camera.y) * Math.min(1, 6 * realDt);
   388→   388→
   389→   389→    this.debugInfo = `wave=${this.wave} enemies=${this.enemies.length} lvl=${this.player.level} hp=${Math.ceil(this.player.hp)}`;
   390→   390→  }

   392→   392→  private updateHints(dt: number): void {
   393→   393→    this.hintT += dt;
   394→   394→    const show = (id: string) => !this.hintsShown.has(id);
   395→   395→    if (this.time > 0.5 && show('move')) {
   396→   396→      this.hintsShown.add('move');
   397→   397→      this.addText(this.player.x, this.player.y - 60, 'WASD / ARROWS — MOVE', '#9adcff', 15, true);
   398→   398→    }
   399→   399→    if (this.time > 5 && show('fire')) {
   400→   400→      this.hintsShown.add('fire');
   401→   401→      this.addText(this.player.x, this.player.y - 60, 'WEAPONS AUTO-FIRE — STAY ALIVE', '#9adcff', 14, true);
   402→   402→    }
   403→   403→    if (this.time > 10 && show('dash')) {
   404→   404→      this.hintsShown.add('dash');
   405→   405→      this.addText(this.player.x, this.player.y - 60, 'SPACE — DASH THROUGH ENEMIES', '#ffd94d', 15, true);
   406→   406→    }
   407→   407→    if (this.player.heat > 55 && show('heat')) {
   408→   408→      this.hintsShown.add('heat');
   409→   409→      this.addText(this.player.x, this.player.y - 60, 'HEAT RISING — DO NOT GET HIT', '#ffb14d', 14, true);
   410→   410→    }
   411→   411→  }

   428→   428→  spawnEnemy(typeId: string, x: number, y: number, elite: boolean): Enemy | null {
   429→   429→    if (this.enemies.length >= MAX_ENEMIES) return null;
   430→   430→    const base = ENEMY_DEFS[typeId];
   431→   431→    if (!base) return null;
   432→   432→    const wave = this.wave;
   433→   433→    const hpScale = WAVE_CONFIG.hpScale(wave);
   434→   434→    const dmgScale = WAVE_CONFIG.dmgScale(wave);
   435→   435→    const spdScale = WAVE_CONFIG.spdScale(wave);
   436→   436→    const eliteMult = elite ? 6 : 1;
   437→   437→    const mod = elite ? pickMod(this.rng) : null;
   438→   438→    const hp = base.hp * hpScale * eliteMult;
   439→   439→    const e: Enemy = {
   440→   440→      eid: this.nextEid(),
   441→   441→      def: base,
   442→   442→      x,
   443→   443→      y,
   444→   444→      vx: 0,
   445→   445→      vy: 0,
   446→   446→      hp,
   447→   447→      maxHp: hp,
   448→   448→      radius: base.radius * (elite ? 1.55 : 1),
   449→   449→      speed: base.speed * spdScale * (mod === 'swift' ? 1.45 : 1),
   450→   450→      damage: base.damage * dmgScale * (elite ? 1.4 : 1),
   451→   451→      xp: base.xp * (elite ? 6 : 1),
   452→   452→      score: base.score * (elite ? 8 : 1),
   453→   453→      elite,
   454→   454→      mod,
   455→   455→      flash: 0,
   456→   456→      t: 0,
   457→   457→      shootTimer: base.shoot ? base.shoot.cd * (0.5 + Math.random()) : 999,
   458→   458→      phase: 0,
   459→   459→      slow: 0,
   460→   460→      slowT: 0,
   461→   461→      freezeT: 0,
   462→   462→      stunT: 0,
   463→   463→      knockX: 0,
   464→   464→      knockY: 0,
   465→   465→      spawnT: 0.45,
   466→   466→      orbHitCd: 0,
   467→   467→      zigPhase: Math.random() * Math.PI * 2,
   468→   468→      rage: 1,
   469→   469→      shieldHits: mod === 'shield' ? 3 : 0,
   470→   470→      hitCd: new Map<number, number>(),
   471→   471→    };
   472→   472→    this.enemies.push(e);
   473→   473→    if (elite) {
   474→   474→      this.spawnParticleBurst(x, y, '#ff9a3d', 12, 160, 3);
   475→   475→    }
   476→   476→    return e;
   477→   477→  }

   479→   479→  damageEnemy(e: Enemy, dmg: number, crit: boolean, source: string, silent = false): void {
   480→   480→    if (e.hp <= 0) return;
   481→   481→    e.hp -= dmg;
   482→   482→    e.flash = 1;
   483→   483→    if (!silent && this.texts.length < MAX_TEXTS) {
   484→   484→      this.addText(
   485→   485→        e.x + (Math.random() - 0.5) * 14,
   486→   486→        e.y - e.radius - 6,
   487→   487→        String(Math.round(dmg)),

   498→   498→  killEnemy(e: Enemy, source: string): void {

   538→   538→        }
   539→   539→      }
   540→   540→    }
   541→   541→    // elite burst mod
   542→   542→    if (e.mod === 'burst') {
   543→   543→      for (let i = 0; i < 8; i++) {
   544→   544→        const a = (i / 8) * Math.PI * 2;

   547→   547→            x: e.x,
   548→   548→            y: e.y,
   549→   549→            vx: Math.cos(a) * 190,
   550→   550→            vy: Math.sin(a) * 190,
